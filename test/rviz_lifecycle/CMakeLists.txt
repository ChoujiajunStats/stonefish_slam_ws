cmake_minimum_required(VERSION 3.16)
project(uw_rviz_lifecycle_regression)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_executable(image_unsubscribe image_unsubscribe.cpp)
ament_target_dependencies(image_unsubscribe rviz_common rviz_default_plugins)
target_link_libraries(image_unsubscribe Qt5::Widgets)
target_compile_options(image_unsubscribe PRIVATE -fsanitize=address -fno-omit-frame-pointer)
target_link_options(image_unsubscribe PRIVATE -fsanitize=address)
