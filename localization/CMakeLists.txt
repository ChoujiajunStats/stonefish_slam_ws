cmake_minimum_required(VERSION 3.16)
project(uw_localization)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem system)
find_library(OPENVINS_LIBRARY ov_msckf_lib REQUIRED)
find_path(OPENVINS_INCLUDE core/VioManager.h PATHS /usr/local/include/open_vins REQUIRED)
add_executable(openvins src/openvins_node.cpp)
target_compile_features(openvins PRIVATE cxx_std_17)
target_compile_options(openvins PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_compile_definitions(openvins PRIVATE ROS_AVAILABLE=0)
target_include_directories(openvins PRIVATE ${OPENVINS_INCLUDE} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(openvins ${OPENVINS_LIBRARY} Eigen3::Eigen ${OpenCV_LIBRARIES} Boost::filesystem Boost::system)
ament_target_dependencies(openvins rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs std_srvs rosgraph_msgs tf2_ros tf2_msgs)
install(TARGETS openvins DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config/ DESTINATION share/${PROJECT_NAME}/config)

# Optional additive image: older M0-M3 images do not require ORB-SLAM3.
if(EXISTS /opt/uw/orbslam3-lock.sha256)
  set(ORB_ROOT /opt/uw_src/ORB_SLAM3)
  find_package(Pangolin REQUIRED)
  add_executable(orbslam3 src/orbslam3_node.cpp)
  target_compile_features(orbslam3 PRIVATE cxx_std_17)
  target_include_directories(orbslam3 PRIVATE ${ORB_ROOT} ${ORB_ROOT}/include ${ORB_ROOT}/include/CameraModels ${ORB_ROOT}/Thirdparty/Sophus ${OpenCV_INCLUDE_DIRS} ${Pangolin_INCLUDE_DIRS})
  target_link_libraries(orbslam3 ${ORB_ROOT}/lib/libORB_SLAM3.so Eigen3::Eigen ${OpenCV_LIBRARIES} ${Pangolin_LIBRARIES})
  ament_target_dependencies(orbslam3 rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros)
  add_executable(orbslam3_atlas_check src/orbslam3_atlas_check.cpp)
  target_compile_features(orbslam3_atlas_check PRIVATE cxx_std_17)
  target_include_directories(orbslam3_atlas_check PRIVATE ${ORB_ROOT} ${ORB_ROOT}/include ${ORB_ROOT}/include/CameraModels ${ORB_ROOT}/Thirdparty/Sophus ${OpenCV_INCLUDE_DIRS} ${Pangolin_INCLUDE_DIRS})
  target_link_libraries(orbslam3_atlas_check ${ORB_ROOT}/lib/libORB_SLAM3.so Eigen3::Eigen ${OpenCV_LIBRARIES} ${Pangolin_LIBRARIES})
  install(TARGETS orbslam3 orbslam3_atlas_check DESTINATION lib/${PROJECT_NAME})
endif()

ament_package()
